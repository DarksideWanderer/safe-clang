cmake_minimum_required(VERSION 3.20)
project(ModuleSafety LANGUAGES C CXX)
find_package(LLVM REQUIRED CONFIG)
find_package(Clang REQUIRED CONFIG)
add_library(ModuleSafety MODULE src/ModuleSafety.cpp src/FlowAnalysis.cpp)
set_property(DIRECTORY APPEND PROPERTY CMAKE_CONFIGURE_DEPENDS src/ModuleSafety.cpp src/FlowAnalysis.cpp src/FlowAnalysis.h src/SafetyContracts.h src/SafetyTraversal.h)
set(MODULE_SAFETY_HASH_INPUT "")
foreach(SOURCE ModuleSafety.cpp FlowAnalysis.cpp FlowAnalysis.h SafetyContracts.h SafetyTraversal.h)
  file(SHA256 "${CMAKE_CURRENT_SOURCE_DIR}/src/${SOURCE}" SOURCE_HASH)
  string(APPEND MODULE_SAFETY_HASH_INPUT "${SOURCE_HASH}")
endforeach()
string(SHA256 MODULE_SAFETY_HASH "${MODULE_SAFETY_HASH_INPUT}")
target_compile_definitions(ModuleSafety PRIVATE MODULE_SAFETY_FINGERPRINT="${MODULE_SAFETY_HASH}")
target_compile_features(ModuleSafety PRIVATE cxx_std_17)
target_include_directories(ModuleSafety SYSTEM PRIVATE ${LLVM_INCLUDE_DIRS} ${CLANG_INCLUDE_DIRS})
target_link_libraries(ModuleSafety PRIVATE clang-cpp LLVM)
set_target_properties(ModuleSafety PROPERTIES PREFIX "")
find_package(Python3 REQUIRED COMPONENTS Interpreter)
enable_testing()
